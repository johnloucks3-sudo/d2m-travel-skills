# 2. IMAGE HIERARCHY (TITAN ART DRIVE -> SNIPER VISION -> PEXELS)

def get_drive_service():
    creds = service_account.Credentials.from_service_account_file(
        Config.SERVICE_ACCOUNT_FILE,
        scopes=['https://www.googleapis.com/auth/drive.readonly']
    )
    return build('drive', 'v3', credentials=creds)

def find_curated_art(port_name: str) -> str:
    """Searches Drive for the specific Titan_Art convention."""
    service = get_drive_service()
    # Convention: Titan_Art_City__Country_ID.png
    search_term = f"Titan_Art_{port_name}"
    query = f"'{Config.ART_FOLDER_ID}' in parents and name contains '{search_term}'"
    try:
        results = service.files().list(q=query, fields="files(id, webContentLink, name)").execute()
        files = results.get('files', [])
        if files:
            logger.info(f"🎨 Thunderbird Art MATCH: {files[0]['name']}")
            return files[0].get('webContentLink')
    except: pass
    return None

def get_best_image(query: str, port_name: str) -> str:
    art = find_curated_art(port_name)
    if art: return art

    logger.info(f"🎯 Sniper Vision searching for: {port_name}")
    search_url = "https://www.googleapis.com/customsearch/v1"
    params = {
        "q": f"{port_name} luxury travel photography {query}",
        "cx": Config.GOOGLE_CX_ID, "key": Config.GOOGLE_SEARCH_API_KEY,
        "searchType": "image", "imgSize": "huge", "num": 1
    }
    try:
        res = requests.get(search_url, params=params).json()
        if 'items' in res: return res['items'][0]['link']
    except: pass

    # Pexels Final Fallback
    headers = {"Authorization": Config.PEXELS_KEY}
    try:
        res = requests.get(f"https://api.pexels.com/v1/search?query={query}&per_page=1", headers=headers).json()
        if 'photos' in res: return res['photos'][0]['src']['large2x']
    except: pass

    return "https://images.unsplash.com/photo-1548574505-5e239809ee19"

# 3. THE 49-COLUMN PDF VACUUM (OCR + GROQ)
# ============================================================================
def process_pdfs():
    creds = service_account.Credentials.from_service_account_file(Config.SERVICE_ACCOUNT_FILE, scopes=['https://www.googleapis.com/auth/spreadsheets', 'https://www.googleapis.com/auth/drive'])
    gc = gspread.authorize(creds)
    doc = gc.open_by_key(Config.SHEET_ID)
    master_ws = doc.worksheet(Config.MASTER_TAB)
    daily_ws = doc.worksheet(Config.DAILY_TAB)

    m_headers = master_ws.get_all_values()[0]
    mh_map = {h.strip(): i for i, h in enumerate(m_headers) if h.strip()}
    d_headers = daily_ws.get_all_values()[0]
    dh_map = {h.strip(): i for i, h in enumerate(d_headers) if h.strip()}

    pdf_files = list(Config.INPUT_DIR.glob("*.pdf"))
    if not pdf_files:
        print("No PDFs found in Downloads/BookingPDFs!")
        return

    for pdf in pdf_files:
        logger.info(f"🔍 Analyzing {pdf.name}...")
        images = convert_from_path(pdf)
        text = "".join([pytesseract.image_to_string(img) for img in images])

        # Groq Extraction
        url = "https://api.groq.com/openai/v1/chat/completions"
        m_prompt = f"Extract all travel data into JSON. Output ONLY JSON. TEXT: {text[:15000]}"
        res = requests.post(url, json={"model": "llama-3.3-70b-versatile", "messages": [{"role": "user", "content": m_prompt}], "response_format": {"type": "json_object"}}, headers={"Authorization": f"Bearer {Config.GROQ_KEY}"}).json()
        data = json.loads(res['choices'][0]['message']['content'])

        # Push to Master
        new_row = ["" for _ in range(len(m_headers))]
        for k, v in data.items():
            if k in mh_map: new_row[mh_map[k]] = str(v)
        master_ws.append_row(new_row)

        # Push Itinerary
        i_prompt = f"Extract cruise ports into JSON schedule with Romance_Short 3-word visual query. TEXT: {text[:15000]}"
        res_i = requests.post(url, json={"model": "llama-3.3-70b-versatile", "messages": [{"role": "user", "content": i_prompt}], "response_format": {"type": "json_object"}}, headers={"Authorization": f"Bearer {Config.GROQ_KEY}"}).json()
        itin = json.loads(res_i['choices'][0]['message']['content']).get("schedule", [])

        if itin:
            rows = []
            for day in itin:
                d_row = ["" for _ in range(len(d_headers))]
                d_row[dh_map['Confirmation_Number']] = data.get('Confirmation_Number', '')
                for k, v in day.items():
                    if k in dh_map: d_row[dh_map[k]] = str(v)
                rows.append(d_row)
            daily_ws.append_rows(rows)

        Config.PROCESSED_DIR.mkdir(parents=True, exist_ok=True)
        pdf.rename(Config.PROCESSED_DIR / pdf.name)
        logger.info(f"📦 Archived {pdf.name}")

def generate_itinerary():
    name = input("Client Name (Exact from Sheet): ").strip()
    logger.info(f"⚡ 🎨 Generating Masterpiece for {name}...")
    creds = service_account.Credentials.from_service_account_file(Config.SERVICE_ACCOUNT_FILE, scopes=['https://www.googleapis.com/auth/spreadsheets'])
    gc = gspread.authorize(creds)
    doc = gc.open_by_key(Config.SHEET_ID)

    m_ws = doc.worksheet(Config.MASTER_TAB)
    m_vals = m_ws.get_all_values()
    m_rec = [dict(zip(m_vals[0], r)) for r in m_vals[1:]]
    master_row = next((r for r in m_rec if name.lower() in str(r.get('Client_Name','')).lower()), None)

    if not master_row: return logger.error(f"Client {name} not found.")

    d_ws = doc.worksheet(Config.DAILY_TAB)
    d_vals = d_ws.get_all_values()
    d_headers = d_vals[0]

    # NEW LOGIC: Capture the exact sheet row number before processing
    d_rec = []
    for i, r in enumerate(d_vals[1:]):
        row_dict = dict(zip(d_headers, r))
        row_dict['__sheet_row__'] = i + 2  # +1 for header, +1 for 1-based indexing
        d_rec.append(row_dict)

    daily_rows = [r for r in d_rec if str(r.get('Confirmation_Number')) == str(master_row.get('Confirmation_Number'))]

    client = OpenAI(api_key=Config.GEMINI_API_KEY, base_url=Config.GEMINI_BASE_URL)

    for i, day in enumerate(daily_rows):
        row_idx = day['__sheet_row__'] # Direct access, no more .index() search

        # Clean Romance Storytelling
        if not day.get('Romance_Narrative') or "luxury sentences" in day.get('Romance_Narrative', '').lower():
            logger.info(f"⚡ ✍ Storytelling for {day['Port_Location']}...")
            prompt = f"Write 3 luxury travel sentences about {day['Port_Location']}. Prose only. No introductory phrases."
            res = client.chat.completions.create(model=Config.GEMINI_MODEL, messages=[{"role":"user", "content":prompt}])
            day['Romance_Narrative'] = res.choices[0].message.content.strip()

            # Update Sheet Row using the pinned index
            d_ws.update_cell(row_idx, d_headers.index('Romance_Narrative') + 1, day['Romance_Narrative'])

        # Sniper Image Fetch
        if not day.get('Image_1_URL'):
            img = get_best_image(day.get('Romance_Short', 'Luxury'), day.get('Port_Location'))
            day['Image_1_URL'] = img
            # Update Sheet Row using the pinned index
            d_ws.update_cell(row_idx, d_headers.index('Image_1_URL') + 1, img)

    # FINAL RENDERING (Template remains identical)
    html_content = Template(HTML_TEMPLATE).render(**master_row, days=daily_rows,
                                                LOGO_URL=Config.LOGO_URL, AGENT_PHOTO_URL=Config.AGENT_PHOTO_URL,
                                                AGENT_NAME=Config.AGENT_NAME, AGENT_PHONE=Config.AGENT_PHONE, AGENT_EMAIL=Config.AGENT_EMAIL)
    Config.OUTPUT_DIR.mkdir(parents=True, exist_ok=True)
    out_path = Config.OUTPUT_DIR / f"MASTERPIECE_{name.replace(' ', '_')}.pdf"
    HTML(string=html_content).write_pdf(out_path)
    logger.info(f"✅ FINAL MASTERPIECE SAVED: {out_path}")
